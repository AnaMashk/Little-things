function getSeason() {
  const m = new Date().getMonth();
  if (m >= 2 && m <= 4) return 'spring';
  if (m >= 5 && m <= 7) return 'summer';
  if (m >= 8 && m <= 10) return 'autumn';
  return 'winter';
}

function compressImage(file, maxDim, quality) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = e => {
      const img = new Image();
      img.onload = () => {
        let w = img.width, h = img.height;
        if (w > h) { if (w > maxDim) { h = Math.round(h * maxDim / w); w = maxDim; } }
        else       { if (h > maxDim) { w = Math.round(w * maxDim / h); h = maxDim; } }
        const canvas = document.createElement('canvas');
        canvas.width = w; canvas.height = h;
        const ctx2 = canvas.getContext('2d');
        ctx2.drawImage(img, 0, 0, w, h);
        try { resolve(canvas.toDataURL('image/jpeg', quality)); }
        catch(err) { reject(err); }
      };
      img.onerror = reject;
      img.src = e.target.result;
    };
    reader.onerror = reject;
    reader.readAsDataURL(file);
  });
}

function formatTime(ts) {
  if (!ts) return '—';
  const d = new Date(ts);
  return String(d.getDate()).padStart(2,'0') + '.' + String(d.getMonth()+1).padStart(2,'0') + ' · ' + String(d.getHours()).padStart(2,'0') + ':' + String(d.getMinutes()).padStart(2,'0');
}
function formatDuration(sec) {
  if (!sec || sec < 0) return '0 сек';
  const h = Math.floor(sec / 3600), m = Math.floor((sec % 3600) / 60), s = sec % 60;
  if (h > 0) return h + ' ч ' + m + ' мин';
  if (m > 0) return m + ' мин ' + s + ' сек';
  return s + ' сек';
}

function getProgressToNext(p) {
  const cur = getLevel(p); const idx = LEVELS.indexOf(cur);
  if (idx === LEVELS.length - 1) return 100;
  const next = LEVELS[idx + 1];
  const prevMax = idx === 0 ? 0 : LEVELS[idx - 1].max + 1;

}

function animateNumber(el, from, to, duration) {
  duration = duration || 900;
  const start = performance.now();
  function tick(now) {
    const t = Math.min(1, (now - start) / duration);
    const eased = 1 - Math.pow(1 - t, 3);
    el.textContent = Math.round(from + (to - from) * eased);
    if (t < 1) requestAnimationFrame(tick);
    else el.textContent = to;
  }
  requestAnimationFrame(tick);
}

function logActivity(action, detail) {
  const now = Date.now();
  if (state.counters[action] !== undefined) state.counters[action]++;
  else state.counters[action] = 1;
  const todayKey = new Date().toISOString().slice(0,10);
  if (state.todayCounters.date !== todayKey) {
    state.todayCounters = { date: todayKey, spin:0, task_done:0, page_opens:0, day_opens:0, settings_opens:0, logins:0 };
  }
  if (action === 'spin') state.todayCounters.spin++;
  if (action === 'task_done') state.todayCounters.task_done++;
  if (action === 'login') state.todayCounters.logins++;
  if (action === 'settings_open') state.todayCounters.settings_opens++;
  if (action === 'day_open') state.todayCounters.day_opens++;
  if (action.indexOf('page_') === 0) state.todayCounters.page_opens++;
  state.activityLog.unshift({ id: now, action: action, detail: detail || '', timestamp: now });
  if (state.activityLog.length > 200) state.activityLog = state.activityLog.slice(0, 200);
  saveState();
}

function subscribeToFirestore() {
  unsubscribeFirestore = onSnapshot(mainDocRef, async (snapshot) => {
    if (snapshot.metadata.hasPendingWrites) return;
    if (snapshot.exists()) {
      const d = snapshot.data() || {};
      const prevPoints = state.points;
      if (d.currentMonth && d.currentMonth !== getCurrentMonthKey()) {
        state.points = 0; state.spinsUsed = {};
        setDoc(mainDocRef, { points:0, currentMonth:getCurrentMonthKey(), spinsUsed:{} }, { merge:true });
      } else { state.points = d.points || 0; }
      state.totalPoints = d.totalPoints || 0;
      state.history = d.history || [];
      state.completedDays = d.completedDays || {};
      state.spinsUsed = d.spinsUsed || {};
      state.dailyVisits = d.dailyVisits || {};
      state.quarantine = d.quarantine || {};
      state.userAdded = d.userAdded || { emotional:[], physical:[], lightDate:[], bigDate:[] };
      state.userRemoved = d.userRemoved || { emotional:[], physical:[], lightDate:[], bigDate:[] };
      state.wishes = d.wishes || [];
      state.notificationsForNastya = d.notificationsForNastya || [];
      state.activityLog = d.activityLog || [];
      state.achievements = d.achievements || [];
      state.lastCelebrationMonth = d.lastCelebrationMonth || null;
      state.vanyaSession = d.vanyaSession || state.vanyaSession;
      state.counters = d.counters || state.counters;
      state.todayCounters = d.todayCounters || state.todayCounters;
      if (d.lastLevelName) { lastLevelName = d.lastLevelName; localStorage.setItem('lastLevelName', lastLevelName); }
      if (d.reachedMilestones) { reachedMilestones = d.reachedMilestones; localStorage.setItem('reachedMilestones', JSON.stringify(reachedMilestones)); }
      if (d.welcomeShown !== undefined) localStorage.setItem('welcomeShown', d.welcomeShown ? 'true' : 'false');
      if (d.dismissed_notifications) localStorage.setItem('dismissed_notifications', JSON.stringify(d.dismissed_notifications));
      if (d.schedules) for (let key in d.schedules) localStorage.setItem(key, d.schedules[key]);
      refreshAll(prevPoints);
    } else { saveState(); }
  }, err => console.error('Firestore sub:', err));
}

function isInQuarantine(taskText, cat) {
  const q = state.quarantine[taskText];
  if (!q) return false;
  return (Date.now() - q.doneAt) / (1000*60*60*24) < (QUARANTINE_DAYS[cat] || 14);
}
function putInQuarantine(taskText, cat) {
  state.quarantine[taskText] = { type:cat, doneAt:Date.now() };
  saveState();
}
function openQuarantineList() {
  const list = document.getElementById('quarantineList');
  const items = [];
  for (let k in state.quarantine) {
    if (state.quarantine.hasOwnProperty(k) && isInQuarantine(k, state.quarantine[k].type)) items.push([k, state.quarantine[k]]);
  }
  if (!items.length) list.innerHTML = '<div class="empty-state">Все задания доступны</div>';
  else {
    list.innerHTML = items.map(pair => {
      const txt = pair[0], q = pair[1];
      const days = QUARANTINE_DAYS[q.type] || 14;
      const left = Math.ceil(days - (Date.now() - q.doneAt)/(1000*60*60*24));
      return '<div class="quarantine-item"><span>'+txt+'</span><span class="days">'+left+' дн.</span></div>';
    }).join('');
  }
  document.getElementById('quarantineOverlay').classList.add('active');
}

function getTasksForCategory(cat) {
  const base = DEFAULT_TASKS[cat] || [];
  const added = state.userAdded[cat] || [];
  const removed = state.userRemoved[cat] || [];
  return base.concat(added).filter(t => removed.indexOf(t) === -1);
}
function getAvailableTasks(cat) {
  const all = getTasksForCategory(cat);
  const available = all.filter(t => !isInQuarantine(t, cat));
  if (available.length === 0) {
    return all.slice().sort((a,b) => (state.quarantine[a]?state.quarantine[a].doneAt:0) - (state.quarantine[b]?state.quarantine[b].doneAt:0)).slice(0,5);
  }
  return available;
}

function generateMonthSchedule(year, month) {
  const key = 'schedule_' + year + '_' + month;
  const saved = localStorage.getItem(key);
  if (saved) return JSON.parse(saved);
  const daysInMonth = new Date(year, month+1, 0).getDate();
  const schedule = {}; const weeks = []; let cur = [];
  for (let d=1; d<=daysInMonth; d++) {
    cur.push(d);
    if (new Date(year,month,d).getDay()===0 || d===daysInMonth) { weeks.push(cur); cur=[]; }
  }
  weeks.forEach(w => {
    const sh = w.slice().sort(() => Math.random()-0.5);
    if (sh.length>=1) schedule[sh[0]]='emotional';
    if (sh.length>=2) schedule[sh[1]]='emotional';
    if (sh.length>=3) schedule[sh[2]]='physical';
    if (sh.length>=4) schedule[sh[3]]='physical';
    for (let i=4; i<sh.length; i++) schedule[sh[i]]='free';
  });
  const midC=[]; for (let d1=13; d1<=Math.min(17,daysInMonth); d1++) midC.push(d1);
  if (midC.length) schedule[midC[Math.floor(Math.random()*midC.length)]]='bigDate';
  const endC=[]; for (let d2=25; d2<=Math.min(30,daysInMonth); d2++) endC.push(d2);
  if (endC.length) schedule[endC[Math.floor(Math.random()*endC.length)]]='lightDate';
  localStorage.setItem(key, JSON.stringify(schedule));
  return schedule;
}
function getDayType(year, month, day) {
  const key = year+'-'+month+'-'+day;
  if (state.completedDays[key] && state.completedDays[key].length>0) return state.completedDays[key][0].type;
  return generateMonthSchedule(year, month)[day] || 'free';
}
function getCompletedEvents(y,m,d) { return state.completedDays[y+'-'+m+'-'+d] || []; }

function renderCalendar() {
  const year = state.currentDate.getFullYear();
  const month = state.currentDate.getMonth();
  document.getElementById('monthName').textContent = monthNames[month];
  document.getElementById('monthYear').textContent = year;
  const firstDay = new Date(year,month,1);
  const daysInMonth = new Date(year,month+1,0).getDate();
  let startDay = firstDay.getDay()-1; if (startDay===-1) startDay=6;
  const lastDow = new Date(year,month,daysInMonth).getDay();
  const trailing = lastDow===0 ? 0 : 7-lastDow;
  const cal = document.getElementById('calendar'); cal.innerHTML = '';
  for (let i=0; i<startDay; i++) { const e = document.createElement('div'); e.className='calendar-day empty'; cal.appendChild(e); }
  const today = new Date();
  for (let d=1; d<=daysInMonth; d++) {
    const el = document.createElement('div'); el.className='calendar-day';
    const events = getCompletedEvents(year,month,d);
    const isDone = events.length>0;
    const type = getDayType(year,month,d);
    if (isDone) {
      const types=[]; events.forEach(ev => { if (types.indexOf(ev.type)===-1) types.push(ev.type); });
      el.classList.add('done-'+types[0]);
    } else { el.classList.add(type); }
    el.dataset.day = d; el.dataset.type = type;
    const num = document.createElement('span'); num.className='day-number'; num.textContent = d; el.appendChild(num);
    if (d===today.getDate() && month===today.getMonth() && year===today.getFullYear()) el.classList.add('today');
    el.onclick = (function(day,t,isD){ return function(){ onDayClick(day,t,isD); }; })(d,type,isDone);
    cal.appendChild(el);
  }
  for (let j=1; j<=trailing; j++) {
    const e2 = document.createElement('div'); e2.className='calendar-day next-month';
    const n2 = document.createElement('span'); n2.className='day-number'; n2.textContent = j; e2.appendChild(n2); cal.appendChild(e2);
  }
}

function openPhotoAttach(onSave) {
  state.pendingPhoto = null;
  const overlay = document.getElementById('photoAttachOverlay');

  const skipBtn = document.getElementById('photoSkipBtn');

  saveBtn.disabled = true;
  input.value = '';
  input.onchange = async (e) => {
    const file = e.target.files && e.target.files[0];
    if (!file) return;
    try {
      const dataUrl = await compressImage(file, PHOTO_MAX_DIM, PHOTO_JPEG_QUALITY);
      state.pendingPhoto = dataUrl;
      previewImg.src = dataUrl;
      preview.classList.add('show');

    state.pendingPhoto = null; preview.classList.remove('show');

  };
  saveBtn.onclick = () => { overlay.classList.remove('active'); onSave(); };
  skipBtn.onclick = () => { state.pendingPhoto = null; overlay.classList.remove('active'); onSave(); };
  overlay.classList.add('active');

function addEventToDay(year, month, day, type, text, points, isInitiative) {
  const key = year+'-'+month+'-'+day;
  if (!state.completedDays[key]) state.completedDays[key] = [];
  const event = { type:type, text:text, points:points, isInitiative:isInitiative, timestamp:Date.now() };
  if (state.pendingPhoto) event.photo = state.pendingPhoto;
  state.pendingPhoto = null;
  state.completedDays[key].push(event);
  addPoints(points);
  putInQuarantine(text, type);
  const dateStr = String(day).padStart(2,'0')+'.'+String(month+1).padStart(2,'0')+'.'+year;
  state.history.unshift({
    date:dateStr, eventDate:new Date(year,month,day).getTime(), title:text,
    category:isInitiative?TYPE_INFO[type].name+' (инициатива)':TYPE_INFO[type].name,
    emoji:isInitiative?'⭐':TYPE_INFO[type].emoji, points:points, photo: event.photo || null
  });
  logActivity('task_done', TYPE_INFO[type].emoji + ' ' + text + ' (+' + points + ' б.)');
  return saveState().then(() => {
    renderCalendar(); updateStats(); checkUpcomingDates();
    renderHistory(); renderLoveTree();
    showToast(TYPE_INFO[type].emoji, '+' + points + ' баллов! ' + text);
    fireBurst(window.innerWidth/2, window.innerHeight/2);
    checkAchievements();
    checkMonthCelebration();
  });
}

      logActivity('wish_activate', '💎 Активировал: ' + w.title + ' (−' + w.cost + ' б.)');
      saveState().then(() => { updateStats(); renderWishes(); showToast('💎','Желание активировано!'); makeConfetti(25); checkAchievements(); });
    };
  });
  garden.querySelectorAll('.flower-actions button').forEach(b => {
    b.onclick = () => {
      const i = parseInt(b.dataset.idx); const action = b.dataset.action; const w = state.wishes[i];
      if (action === 'edit') {
        const nv = prompt('Изменить желание:', w.title);
        if (nv && nv.trim()) { state.wishes[i].title = nv.trim(); saveState().then(() => { renderWishes(); showToast('✏️','Изменено'); }); }
      } else if (action === 'delete') {
        if (!confirm('Удалить желание?')) return;
        state.wishes.splice(i, 1);
        saveState().then(() => { renderWishes(); showToast('🗑','Удалено'); });
      }
    };
  });
}

function getAchievementStats() {
  let emotionalDone = 0, physicalDone = 0, bigDateDone = 0, dateDone = 0, initiativeDone = 0, totalDone = 0;
  for (let k in state.completedDays) {
    if (!state.completedDays.hasOwnProperty(k)) continue;
    state.completedDays[k].forEach(e => {
      totalDone++;
      if (e.type === 'emotional') emotionalDone++;
      if (e.type === 'physical') physicalDone++;
      if (e.type === 'bigDate') bigDateDone++;
      if (e.type === 'lightDate' || e.type === 'bigDate') dateDone++;
      if (e.isInitiative) initiativeDone++;
    });
  }
  return {
    totalDone, emotionalDone, physicalDone, bigDateDone, dateDone, initiativeDone,
    wishesCount: (state.wishes || []).length,
    wishesActivated: (state.wishes || []).filter(w => w.activated).length,
    totalPoints: state.totalPoints || 0
  };
}
function checkAchievements() {
  const s = getAchievementStats();

  ACHIEVEMENTS.forEach(a => {
    if (state.achievements.indexOf(a.id) === -1 && a.check(s)) {

      setTimeout(() => { showToast(a.emoji, 'Новое достижение: ' + a.name + '!'); }, i * 700);
    });
  }
}

function checkMonthCelebration() {
  if (state.viewingAs !== 'vanya') return;
  const now = new Date();
  const prevMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
  const prevMonthKey = prevMonth.getFullYear() + '-' + prevMonth.getMonth();
  if (state.lastCelebrationMonth === prevMonthKey) return;
  let countPrevMonth = 0;
  for (let k in state.completedDays) {
    if (!state.completedDays.hasOwnProperty(k)) continue;
    const parts = k.split('-');
    const y = parseInt(parts[0]); const m = parseInt(parts[1]);
    if (y === prevMonth.getFullYear() && m === prevMonth.getMonth()) countPrevMonth += state.completedDays[k].length;
  }
  if (countPrevMonth >= MONTH_CELEBRATION_DAYS) {
    state.lastCelebrationMonth = prevMonthKey;
    state.points += MONTH_CELEBRATION_BONUS;
    state.totalPoints += MONTH_CELEBRATION_BONUS;
    const mc = document.getElementById('monthCelebration');
    document.getElementById('mcTitle').textContent = 'Восхитительно!';
    document.getElementById('mcText').innerHTML = 'Ты выполнил <b>' + countPrevMonth + '</b> дел за прошлый месяц!<br>Настя гордится тобой.';

    mc.classList.add('active');
    saveState();
    setTimeout(() => makeConfetti(60), 400);
  } else if (!state.lastCelebrationMonth) {
    state.lastCelebrationMonth = prevMonthKey;
    saveState();
  }
}

let wheelItems = [], currentAngle = 0, spinVelocity = 0, isSpinning = false, animId = null, wheelMode = 'task';
let canvas, ctx;
function resizeCanvas() {
  if (!canvas) { canvas = document.getElementById('wheel'); ctx = canvas.getContext('2d'); }
  if (!canvas || !ctx) return;
  const size = Math.min(window.innerWidth * 0.78, 340);
  const dpr = window.devicePixelRatio || 1;
  canvas.width = size * dpr; canvas.height = size * dpr;
  canvas.style.width = size + 'px'; canvas.style.height = size + 'px';
  ctx.setTransform(1,0,0,1,0,0); ctx.scale(dpr,dpr);
  drawWheel();
}
function lighten(hex, amount) {
  const r = parseInt(hex.slice(1,3),16), g = parseInt(hex.slice(3,5),16), b = parseInt(hex.slice(5,7),16);
  return 'rgb('+Math.min(255,Math.round(r+(255-r)*amount))+','+Math.min(255,Math.round(g+(255-g)*amount))+','+Math.min(255,Math.round(b+(255-b)*amount))+')';
}
function darken(hex, amount) {
  const r = parseInt(hex.slice(1,3),16), g = parseInt(hex.slice(3,5),16), b = parseInt(hex.slice(5,7),16);
  return 'rgb('+Math.max(0,Math.round(r*(1-amount)))+','+Math.max(0,Math.round(g*(1-amount)))+','+Math.max(0,Math.round(b*(1-amount)))+')';
}
function drawWheel() {
  if (!wheelItems.length || !ctx) return;
  const size = parseInt(canvas.style.width);
  const cx = size/2, cy = size/2; const r = cx - 24;
  const step = (Math.PI*2)/wheelItems.length;
  ctx.clearRect(0,0,size,size);
  wheelItems.forEach((item, i) => {
    const start = currentAngle + i*step, end = start + step;
    ctx.beginPath(); ctx.moveTo(cx,cy); ctx.arc(cx,cy,r,start,end); ctx.closePath();
    const baseColor = (item.type && TYPE_INFO[item.type]) ? TYPE_INFO[item.type].color : ['#dbeaf3','#e9dff3','#f8e5e9','#f4ead0'][i%4];
    const grad = ctx.createRadialGradient(cx, cy, r*0.15, cx, cy, r);
    grad.addColorStop(0, lighten(baseColor, 0.7));
    grad.addColorStop(0.5, lighten(baseColor, 0.3));
    grad.addColorStop(1, darken(baseColor, 0.05));
    ctx.fillStyle = grad; ctx.fill();
    ctx.strokeStyle = "rgba(232,200,138,0.75)"; ctx.lineWidth = 1.2; ctx.stroke();
  });
  ctx.beginPath(); ctx.arc(cx,cy,r,0,Math.PI*2);
  ctx.strokeStyle = "rgba(58,46,42,0.12)"; ctx.lineWidth = 1; ctx.stroke();
  ctx.beginPath(); ctx.arc(cx,cy,30,0,Math.PI*2);
  ctx.fillStyle = "#FDFBF7"; ctx.fill();
  ctx.strokeStyle = "rgba(212,168,90,0.5)"; ctx.lineWidth = 1; ctx.stroke();
}

function getSpinsKeyForTarget() {
  if (state.currentDate && state.currentDay) return 'target-'+state.currentDate.getFullYear()+'-'+state.currentDate.getMonth()+'-'+state.currentDay;
  const d = new Date();
  return 'target-'+d.getFullYear()+'-'+d.getMonth()+'-'+d.getDate();
}
function getSpinsUsed() { return state.spinsUsed[getSpinsKeyForTarget()] || 0; }
function getNextSpinPrice() { const used = getSpinsUsed(); return SPIN_PRICES[Math.min(used, SPIN_PRICES.length - 1)]; }
function updateSpinsUI() {
  if (wheelMode === 'type') {
    document.getElementById('spinBtn').disabled = false;
    document.getElementById('spinBtn').textContent = 'Крутить колесо';
    document.getElementById('spinsLeft').textContent = 'Выбор категории бесплатный';
    return;
  }
  const price = getNextSpinPrice();
  const btn = document.getElementById('spinBtn');
  const info = document.getElementById('spinsLeft');
  const used = getSpinsUsed();
  const targetDay = state.currentDay;
  if (used === 0) { btn.textContent = 'Крутить колесо'; btn.disabled = false; info.textContent = 'Первая попытка для ' + targetDay + '-го — бесплатно ✨'; }
  else if (state.points >= price) { btn.textContent = 'Крутить ещё (−' + price + ' б.)'; btn.disabled = false; info.textContent = 'Перекрут: −' + price + ' б.'; }
  else { btn.textContent = 'Не хватает ' + price + ' б.'; btn.disabled = true; info.textContent = 'Нужно ' + price + ' баллов'; }
}
function animate() {
  if (spinVelocity > 0.0015) {
    currentAngle += spinVelocity; spinVelocity *= 0.991;
    drawWheel(); animId = requestAnimationFrame(animate);
  } else { isSpinning = false; spinVelocity = 0; cancelAnimationFrame(animId); determineWinner(); }
}
function determineWinner() {
  const n = wheelItems.length; const step = (Math.PI*2)/n;
  const norm = ((Math.PI*2) - (currentAngle % (Math.PI*2))) % (Math.PI*2);
  const pa = (norm + Math.PI*1.5) % (Math.PI*2);
  let idx = Math.floor(pa/step) % n; if (idx < 0) idx = 0;
  showResult(wheelItems[idx]);
}
function showResult(item) {
  document.getElementById('resultText').textContent = item.text;
  document.getElementById('resultBox').style.display = 'block';
  window._currentItem = item;
}
function doSpin() {
  if (isSpinning) return;
  if (wheelMode === 'type') { isSpinning = true; spinVelocity = 0.25 + Math.random()*0.15; animate(); return; }
  const price = getNextSpinPrice();
  if (price > 0) {
    if (state.points < price) { showToast('💔', 'Недостаточно баллов'); return; }
    state.points -= price; updateStats();
    showToast('🔄', '−' + price + ' баллов за перекрут');
  }
  const k = getSpinsKeyForTarget();
  state.spinsUsed[k] = (state.spinsUsed[k] || 0) + 1;
  saveState();
  isSpinning = true;
  document.getElementById('resultBox').style.display = 'none';
  spinVelocity = 0.28 + Math.random()*0.15;
  animate();
  updateSpinsUI();
  logActivity('spin', '🎡 Крутил колесо (' + (price > 0 ? '−' + price + ' б.' : 'бесплатно') + ')');
}

function renderActionLists() {
  ['emotional','physical','lightDate','bigDate'].forEach(cat => {
    const ul = document.getElementById('list-'+cat);
    if (!ul) return;
    const tasks = getTasksForCategory(cat);
    ul.innerHTML = tasks.map(t => '<li><span class="txt">'+t+'</span><button class="del-btn" data-cat="'+cat+'" data-text="'+encodeURIComponent(t)+'" type="button">✕</button></li>').join('');
  });
  document.querySelectorAll('.del-btn').forEach(b => {
    b.onclick = () => {
      const cat = b.dataset.cat; const txt = decodeURIComponent(b.dataset.text);
      if (!confirm('Удалить это действие?')) return;
      if (DEFAULT_TASKS[cat].indexOf(txt) !== -1) {
        if (!state.userRemoved[cat]) state.userRemoved[cat] = [];
        if (state.userRemoved[cat].indexOf(txt) === -1) state.userRemoved[cat].push(txt);
      } else { state.userAdded[cat] = (state.userAdded[cat]||[]).filter(x => x !== txt); }
      saveState().then(() => { renderActionLists(); showToast('✓','Действие удалено'); });
    };
  });
}
